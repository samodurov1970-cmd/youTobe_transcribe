WITH params AS (SELECT $1::jsonb AS p),
flags AS (
  SELECT
    COALESCE((SELECT (p->>'is_rate_limited')::boolean FROM params), false) AS rate_limited,
    COALESCE((SELECT (p->>'is_auth_error')::boolean FROM params), false) AS auth_error
),
stop_flag AS (
  SELECT (rate_limited OR auth_error) AS flag FROM flags
),
mark AS (
  UPDATE videos SET is_processed = true, processed_at = NOW()
  WHERE video_id = (SELECT p->>'video_id' FROM params)
    AND (SELECT (p->>'mark_processed')::boolean FROM params) = true
    AND (SELECT flag FROM stop_flag) = false
  RETURNING video_id
),
ins_file AS (
  INSERT INTO video_files (
    file_name, video_id, video_url, file_content,
    channel_name, channel_topic, status, is_processed, last_error
  )
  SELECT
    (SELECT p->>'file_name' FROM params),
    (SELECT p->>'video_id' FROM params),
    (SELECT p->>'video_url' FROM params),
    CASE
      WHEN (SELECT flag FROM stop_flag) THEN COALESCE((SELECT p->>'content' FROM params), '')
      WHEN (SELECT (p->>'is_error')::boolean FROM params) THEN ''
      ELSE (SELECT p->>'content' FROM params)
    END,
    (SELECT p->>'channel_name' FROM params),
    (SELECT p->>'channel_topic' FROM params),
    CASE
      WHEN (SELECT rate_limited FROM flags) THEN 'rate_limited'
      WHEN (SELECT auth_error FROM flags) THEN 'auth_error'
      WHEN (SELECT p->>'status' FROM params) = 'too_long' THEN 'too_long'
      WHEN (SELECT (p->>'is_error')::boolean FROM params) THEN 'error'
      ELSE 'pending'
    END,
    CASE
      WHEN (SELECT flag FROM stop_flag) THEN false
      ELSE (SELECT (p->>'is_error')::boolean FROM params)
    END,
    CASE
      WHEN (SELECT flag FROM stop_flag) THEN NULL
      WHEN (SELECT (p->>'is_error')::boolean FROM params) THEN (SELECT p->>'error_message' FROM params)
      ELSE NULL
    END
  WHERE (SELECT (p->>'mark_processed')::boolean FROM params) = true
    AND (SELECT flag FROM stop_flag) = false
  ON CONFLICT (video_id) DO UPDATE SET
    file_name = EXCLUDED.file_name,
    video_url = EXCLUDED.video_url,
    channel_name = EXCLUDED.channel_name,
    channel_topic = EXCLUDED.channel_topic,
    status = EXCLUDED.status,
    is_processed = EXCLUDED.is_processed,
    last_error = EXCLUDED.last_error,
    file_content = EXCLUDED.file_content
  RETURNING video_id
)
